三、全书统一约定（动画与代码严格一致）

  • Clarke 采用幅值不变形式（系数 2/3）；α 轴与 A 相绕组轴线重合，β 轴超前 α 90°。
  • 正方向：逆时针为正。A→B→C 正相序时电流空间矢量逆时针旋转，电角度 θe 增大，对应「正转」。
  • Park：id = iα·cosθ + iβ·sinθ；iq = −iα·sinθ + iβ·cosθ；复数写法 i_dq = i_αβ·e^(−jθ)。反 Park：vα = vd·cosθ − vq·sinθ；vβ = vd·sinθ + vq·cosθ。
  • d 轴对准永磁体磁链方向，q 轴超前 d 轴 90°；iq > 0 产生正向电磁转矩。
  • 相电流正方向：从逆变器流入电机为正。
  • 角度格式 uint16：16384 = 90°，32768 = 180°，加法溢出即自动取模 2π。

/* ===== 全局 FOC 数据结构（模块 A 先出现的字段，后续章节扩充同一结构体） ===== */
typedef int16_t q15_t;                 /* Q15：[-1, 1)                         */

typedef struct {
    uint16_t adcIa, adcIb;             /* 相电流 ADC 原始码值                   */
    q15_t    ia, ib, ic;               /* 相电流 标幺 Q15，基值 FOC_I_BASE      */
    q15_t    iAlpha, iBeta;            /* αβ 电流                               */
    q15_t    id, iq;                   /* dq 电流                               */
    q15_t    idRef, iqRef;             /* dq 电流给定                           */
    q15_t    vd, vq, vAlpha, vBeta;    /* 电压，基值 FOC_V_BASE                 */
    q15_t    vdcPu;                    /* 母线电压标幺                          */
    uint16_t thetaE;                   /* 电角度 0..65535 ↔ 0..2π               */
    uint16_t thetaM;                   /* 机械角度（旋变）                      */
    int16_t  thetaEIncPerTs;           /* 每 Ts 电角度增量（uint16 单位）       */
    q15_t    sinTh, cosTh;
    int32_t  spdRpmFilt;               /* 机械转速 rpm（滤波后）                */
    uint16_t thetaEInvPark;            /* 反 Park 用补偿后电角度（A03）         */
} FOC_tst;

extern FOC_tst gFoc;

/* ===== 标定量：前缀 Cal_，通过 XCP 在 CANape 中在线修改 ===== */

extern volatile int16_t Cal_IdKp_q12, Cal_IqKp_q12;           /* 电流环 Kp     */
extern volatile int16_t Cal_IdKiTs_q15, Cal_IqKiTs_q15;       /* 电流环 Ki·Ts  */
