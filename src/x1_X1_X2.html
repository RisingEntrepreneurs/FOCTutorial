第四部分：嵌入式实现步骤

  1. 中断内按 1→10 顺序执行（B03 中 FOC_FastLoop_Isr）。
  2. 每一级结果写入 gFoc，XCP 以主中断事件同时采集整条链。
  3. 手算复核：任选一个时刻，用记录的 ia、ib、θ 手算 id、iq 与 duty，与记录值比对（PIL / 台架一致性检查）。

第五部分：仿真与波形预期

  • 正常：iq 测量为直线，实际力矩恒定；duty 为马鞍波。
  • 每种扰动在链路上的「首次出现位置」就是 E04 定位流程的依据。

void FOC_FastLoop_Isr(void)
{

    FOC_AngleUpdate_Isr();                             /* 4  A03 / C09 */
    FOC_Park_Isr();                                    /* 5  A04 */

    FOC_CurrentLoop_Isr();                             /* 7、8  C05（含前馈与 d 轴优先限幅）*/
    FOC_InvPark_Isr();                                 /* 9  C03 */
    FOC_Svpwm_Isr();                                   /* 10 C04 → Pwm_WriteDuty 写影子寄存器 */
    FOC_FastDiag_Isr();                                /*    C12 */
}

第三部分：核心原理与公式

维度单精度浮点 float32定点 Q15 / Q31
动态范围极大，基本无需担心溢出受基值限制，需逐点分析
精度相对精度 ~6×10⁻⁸（24 位尾数），绝对精度随数值增大变差绝对精度恒定（Q15：3×10⁻⁵ pu）
典型陷阱大数加小数丢失（角度累加、积分器）；与 0 比较；NaN 传播回绕溢出、截断累积、格式混用、abs(−32768)
执行时间有 FPU 时乘加快；除法、sqrt、三角函数较慢；中断压栈开销整数乘加快；三角函数查表
可移植性依赖 FPU任意 MCU
标定 A2LFLOAT32_IEEE，物理值直观整数 + 换算公式
适合观测器、慢任务、参数计算、需要大动态范围的中间量采样换算、PWM 计算、角度（uint）、需要确定性与位精确比对的环节
【定点实现】本教程案例的基值修正（C07 / C12 中暴露的问题） 原设计：I_BASE = 50 A，采样量程 ±50 A → 无法表示 60 A 电流极限，软件过流阈值无裕量 推荐修正 【案例假设】：采样量程 ±80 A（硬件改增益），I_BASE = 80 A 每码 = 160 A/4096 = 0.039 A；Q15 每码 = 0.039 × 32768/80 = 16 LSB（同样可用移位！） 电流极限 60 A → 24576；软件过流 70 A → 28672，硬件过流 75 A 代价：电流分辨率从 0.024 A 变为 0.039 A；所有标幺参数重算（Kp_pu = Kp·I_BASE/V_BASE） float 角度累加精度：θ 不取模时 ULP(θ) = 2^(⌊log₂θ⌋ − 23)，当 ULP 接近每步增量 Δθ 时角度严重量化
动画在演示什么：用 float32 累加电角度 θ += ωe·Ts 且不取模时，θ 越来越大，float 的最小间隔 ULP 随之增大，每步增量被量化。曲线为每步增量的相对误差随运行时间的变化；对比 uint32 相位累加器（A06）误差恒为极小常数。
拖动滑块：电频率越低（Δθ 越小），误差越早变严重——低速开环拖动最先出问题。
正常：uint32 / 取模 float。异常：不取模 float 运行数分钟后误差达到百分之几十。

可直接复制的定点工具头文件（foc_fix.h）

/* =====================================================================
 * foc_fix.h  —— FOC 定点工具（本教程统一版本）
 * 约定：q15_t = Q15 [-1, 1)；q12_t = Q12 [-8, 8)；q30 状态用 int32
 *       负数右移按算术右移（Arm GCC / IAR / GHS 均如此，需在编码规范中声明）
 * ===================================================================== */
#ifndef FOC_FIX_H
#define FOC_FIX_H
#include <stdint.h>

typedef int16_t q15_t;
typedef int16_t q12_t;
typedef int32_t q30_t;

#define Q15_ONE        ((int32_t)32767)
#define Q15_MIN_       ((int32_t)-32768)
#define Q15_INV_SQRT3  ((q15_t)18919)     /* 1/√3   */
#define Q15_SQRT3_2    ((q15_t)28378)     /* √3/2   */
#define Q15_2PI_INV    ((q15_t)5215)      /* 1/(2π) */

/* 编译期常量转换（仅用于初始化，勿在中断中对变量使用）*/
#define Q15_C(x)  ((q15_t)(((x) >= 0.99997f) ? 32767 : (((x) <= -1.0f) ? -32768 : \
                   (int32_t)((x) * 32768.0f + (((x) >= 0.0f) ? 0.5f : -0.5f)))))
#define Q12_C(x)  ((q12_t)((int32_t)((x) * 4096.0f + (((x) >= 0.0f) ? 0.5f : -0.5f))))

static inline q15_t Q15_Sat(int32_t x)
{ return (q15_t)((x > Q15_ONE) ? Q15_ONE : ((x < Q15_MIN_) ? Q15_MIN_ : x)); }

static inline int32_t Q31_Sat64(int64_t x)
{ return (int32_t)((x > INT32_MAX) ? INT32_MAX : ((x < INT32_MIN) ? INT32_MIN : x)); }

static inline q15_t Q15_Add(q15_t a, q15_t b)   { return Q15_Sat((int32_t)a + b); }
static inline q15_t Q15_Sub(q15_t a, q15_t b)   { return Q15_Sat((int32_t)a - b); }
static inline q15_t Q15_Neg(q15_t a)            { return Q15_Sat(-(int32_t)a); }       /* -(-32768) 饱和 */
static inline q15_t Q15_Abs(q15_t a)            { return (a < 0) ? Q15_Neg(a) : a; }  /* abs(-32768) = 32767 */
static inline q15_t Q15_Mul(q15_t a, q15_t b)   { return Q15_Sat(((int32_t)a * b) >> 15); }
static inline q15_t Q15_MulR(q15_t a, q15_t b)  { return Q15_Sat(((int32_t)a * b + (1 << 14)) >> 15); }
static inline q15_t Q15_MulQ12(q15_t x, q12_t k){ return Q15_Sat(((int32_t)x * k) >> 12); }
static inline q15_t Q15_MAC2(q15_t a, q15_t b, q15_t c, q15_t d)
{ return Q15_Sat(((int32_t)a * b + (int32_t)c * d) >> 15); }               /* a·b + c·d */
static inline q15_t Q15_MSC2(q15_t a, q15_t b, q15_t c, q15_t d)
{ return Q15_Sat(((int32_t)a * b - (int32_t)c * d) >> 15); }               /* a·b − c·d */

/* 一阶低通：y += k·(x − y)，状态 Q30 防截断死区 */
static inline q15_t Q15_Lpf(q30_t *y, q15_t x, q15_t k)
{
    int64_t e = ((int64_t)x << 15) - *y;
    *y = Q31_Sat64((int64_t)*y + ((e * k) >> 15));
    return (q15_t)(*y >> 15);
}

/* 角度：uint16 0..65535 ↔ 0..2π；差值自动处理过零 */
static inline int16_t Ang_Diff(uint16_t a, uint16_t b) { return (int16_t)(uint16_t)(a - b); }

/* 整数开方：输入 Q30 非负 → 输出 Q15 */
static inline q15_t Q15_SqrtQ30(int32_t xQ30)
{
    uint32_t x = (xQ30 > 0) ? (uint32_t)xQ30 : 0u, r = 0u, b = 1uL << 30;
    while (b > x) { b >>= 2; }
    while (b != 0u) { if (x >= r + b) { x -= r + b; r = (r >> 1) + b; } else { r >>= 1; } b >>= 2; }
    return (q15_t)((r > 32767u) ? 32767u : r);
}
#endif /* FOC_FIX_H */

第六部分：EPS 案例实现

/* float 版 PI（慢任务 / 观测器适用），含抗饱和；与定点版 PI_Run（A08）行为一致 */
typedef struct { float kp, kiTs, integ, outMax, outMin; } PIF_tst;
static inline float PIF_Run(PIF_tst *pi, float e)
{
    float iNew = pi->integ + pi->kiTs * e;
    if (iNew > pi->outMax) { iNew = pi->outMax; }
    if (iNew < pi->outMin) { iNew = pi->outMin; }
    float u = pi->kp * e + iNew;
    if (u > pi->outMax)      { u = pi->outMax; if (e < 0.0f) { pi->integ = iNew; } }
    else if (u < pi->outMin) { u = pi->outMin; if (e > 0.0f) { pi->integ = iNew; } }
    else                     { pi->integ = iNew; }
    return u;
}
