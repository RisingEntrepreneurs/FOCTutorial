第六部分：EPS 案例实现

# 【伪代码 / Python】参数表 → C 初始化代码（定点列自动计算）
import csv
BASE = {'I': 50.0, 'V': 16.0, 'Ts': 50e-6}
def kp_q12(kp):   return round(kp * BASE['I'] / BASE['V'] * 4096)
def kits_q15(ki): return round(ki * BASE['Ts'] * BASE['I'] / BASE['V'] * 32768)

with open('params.csv', encoding='utf-8') as f, open('cal_init.c', 'w') as out:
    for row in csv.DictReader(f):                    # 列：name, phys, kind
        v = float(row['phys'])
        q = kp_q12(v) if row['kind'] == 'kp' else kits_q15(v) if row['kind'] == 'ki' else v
        out.write(f"CALIB volatile int16_t {row['name']} = {q};   /* phys = {v} */\n")
